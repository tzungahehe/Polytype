// Hàm phân tích cú pháp chuỗi CSV thành mảng object function parseCSV(text) { const lines = text.trim().split('\n'); if (lines.length <= 1) return []; const headers = lines[0].split(',').map(h => h.trim()); const words = []; // Đọc từng dòng từ vựng for (let i = 1; i < lines.length; i++) { const line = lines[i].trim(); if (!line) continue; // Regex tách dấu phẩy thông minh (bỏ qua dấu phẩy nằm trong ngoặc kép) const values = line.match(/(".*?"|[^",\s]+)(?=\s*,|\s*$)/g) || line.split(','); const row = {}; headers.forEach((header, index) => { let val = values[index] ? values[index].trim() : ''; // Bỏ dấu ngoặc kép bọc ngoài nếu có if (val.startsWith('"') && val.endsWith('"')) { val = val.slice(1, -1); } row[header] = val; }); if (row.word) { words.push(row); } } return words; } // Tải và chuyển đổi file CSV của ngôn ngữ được chọn async function switchLanguage(langId) { const langConfig = languagesList.find(l => l.id === langId); if (!langConfig) return; if (!loadedVocabCache[langId]) { try { const res = await fetch(langConfig.file); if (!res.ok) throw new Error(`Không thể nạp ${langConfig.file}`); const csvText = await res.text(); loadedVocabCache[langId] = parseCSV(csvText); // Parse CSV thành mảng dữ liệu } catch (err) { console.error(`Lỗi nạp file CSV [${langId}]:`, err); alert(`Không tìm thấy file ${langConfig.file}!`); return; } } currentWords = loadedVocabCache[langId] || []; currentLayout = langConfig.keyboardType || 'qwerty'; currentWordIndex = 0; userInput = ''; totalTyped = 0; correctTyped = 0; startTime = null; renderVirtualKeyboard(currentLayout); displayCurrentWord(); }
